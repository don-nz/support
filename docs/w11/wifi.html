WiFi

Windows 11 Wi-Fi reset & cleanup

Run in PowerShell as Administrator. Clears saved profiles, resets the network stack and removes the Wi-Fi adapter so the driver reinstalls on reboot.

Reset-WiFi.ps1 
# Requires Administrator privileges
if (-not ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "ERROR: Please run this script as an Administrator!" -ForegroundColor Red
    Exit
}

Write-Host "==========================================" -ForegroundColor Cyan
Write-Host "    WINDOWS 11 WI-FI RESET & CLEANUP     " -ForegroundColor Cyan
Write-Host "==========================================" -ForegroundColor Cyan

# 1. Remove all saved Wi-Fi profiles
Write-Host "`n[1/5] Deleting saved Wi-Fi profiles..." -ForegroundColor Yellow
netsh wlan delete profile name=* | Out-Null
Write-Host "  -> Saved Wi-Fi profiles cleared." -ForegroundColor Green

# 2. Reset IP Stack, Winsock, and Flush DNS
Write-Host "`n[2/5] Resetting TCP/IP stack and flushing DNS..." -ForegroundColor Yellow
ipconfig /release | Out-Null
ipconfig /flushdns | Out-Null
netsh winsock reset | Out-Null
netsh int ip reset | Out-Null
Write-Host "  -> Network stack reset complete." -ForegroundColor Green

# 3. Disable Power Savings on Wi-Fi Adapter (Prevents disconnects)
Write-Host "`n[3/5] Disabling power-saving management on Wi-Fi adapters..." -ForegroundColor Yellow
$wifiNetAdapters = Get-NetAdapter -Physical -ErrorAction SilentlyContinue | Where-Object { $_.MediaType -eq "Native 802.11" -or $_.Name -like "*Wi-Fi*" -or $_.InterfaceDescription -like "*Wireless*" -or $_.InterfaceDescription -like "*Wi-Fi*" }
if ($wifiNetAdapters) {
    foreach ($adapter in $wifiNetAdapters) {
        Set-NetAdapterPowerManagement -Name $adapter.Name -AllowComputerToTurnOffDevice Never -ErrorAction SilentlyContinue
        Write-Host "  -> Power saving disabled for: $($adapter.Name)" -ForegroundColor Green
    }
} else {
    Write-Host "  -> No active Wi-Fi network interface detected for power tweak." -ForegroundColor DarkGray
}

# 4. Remove Wi-Fi PNP Device & Driver Registration
Write-Host "`n[4/5] Locating and removing Wi-Fi adapter device..." -ForegroundColor Yellow
$pnpAdapters = Get-PnpDevice -Class "Net" -ErrorAction SilentlyContinue | Where-Object { $_.FriendlyName -like "*Wi-Fi*" -or $_.FriendlyName -like "*Wireless*" -or $_.FriendlyName -like "*802.11*" }

if ($pnpAdapters) {
    foreach ($dev in $pnpAdapters) {
        Write-Host "  -> Found: $($dev.FriendlyName) [$($dev.InstanceId)]" -ForegroundColor Cyan
        pnputil /remove-device $dev.InstanceId | Out-Null
        Write-Host "  -> Device successfully removed." -ForegroundColor Green
    }
} else {
    Write-Host "  -> No PNP Wi-Fi adapter found to remove." -ForegroundColor Red
}

# 5. Prompt for Reboot
Write-Host "`n[5/5] Process finished!" -ForegroundColor Cyan
Write-Host "--------------------------------------------------" -ForegroundColor DarkGray
$reboot = Read-Host "A system restart is required to reinstall the driver and apply changes. Restart now? (Y/N)"
if ($reboot -eq "Y" -or $reboot -eq "y") {
    Write-Host "Restarting system in 5 seconds..." -ForegroundColor Yellow
    Start-Sleep -Seconds 5
    Restart-Computer -Force
} else {
    Write-Host "Please restart the computer manually before testing Wi-Fi." -ForegroundColor Yellow
}
